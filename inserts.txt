-- CONSULTAS DE TESTE

-- Usuarios sem postagens
SELECT u.*
FROM usuario u
LEFT JOIN postagem p ON p.idusuario = u.idusuario
WHERE p.idpostagem IS NULL;

-- Usuarios sem seguidores
SELECT u.*
FROM usuario u
LEFT JOIN seguir s ON s.idseguindo = u.idusuario
WHERE s.idseguidor IS NULL;

-- Usuarios que nao seguem ninguem
SELECT u.*
FROM usuario u
LEFT JOIN seguir s ON s.idseguidor = u.idusuario
WHERE s.idseguindo IS NULL;

-- Quantidade de seguidores por usuario
SELECT u.idusuario, u.nome, COUNT(s.idseguidor) AS total
FROM usuario u
LEFT JOIN seguir s ON s.idseguindo = u.idusuario
GROUP BY u.idusuario, u.nome
ORDER BY total DESC;

-- Quantidade de postagens por usuario
SELECT u.idusuario, u.nome, COUNT(p.idpostagem) AS total
FROM usuario u
LEFT JOIN postagem p ON p.idusuario = u.idusuario
GROUP BY u.idusuario, u.nome
ORDER BY total DESC;

-- Postagens sem curtidas
SELECT p.*
FROM postagem p
LEFT JOIN curtida c ON c.idpostagem = p.idpostagem
WHERE c.idpostagem IS NULL;

-- Postagens sem comentarios
SELECT p.*
FROM postagem p
LEFT JOIN comentario c ON c.idpostagem = p.idpostagem
WHERE c.idcomentario IS NULL;

-- Total de curtidas por postagem
SELECT p.idpostagem, p.texto, COUNT(c.idusuario) AS total
FROM postagem p
LEFT JOIN curtida c ON c.idpostagem = p.idpostagem
GROUP BY p.idpostagem, p.texto
ORDER BY total DESC;

-- Total de comentarios por postagem
SELECT p.idpostagem, p.texto, COUNT(c.idcomentario) AS total
FROM postagem p
LEFT JOIN comentario c ON c.idpostagem = p.idpostagem
GROUP BY p.idpostagem, p.texto
ORDER BY total DESC;

-- Feed do usuario 1
SELECT p.idpostagem, p.texto, p.data_hora, u.nome AS autor
FROM seguir s
JOIN postagem p ON p.idusuario = s.idseguindo
JOIN usuario u ON u.idusuario = p.idusuario
WHERE s.idseguidor = 1
ORDER BY p.data_hora DESC;

-- Relacionamentos reciprocos
SELECT s1.idseguidor, s1.idseguindo
FROM seguir s1
JOIN seguir s2
ON s1.idseguidor = s2.idseguindo
AND s1.idseguindo = s2.idseguidor
WHERE s1.idseguidor < s1.idseguindo;
